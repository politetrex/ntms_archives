#include<bits/stdc++.h>
using namespace std;
int Log[200005], a[100005], f[30][100005], n, m;
void build(int n)
{
    Log[1]=0;
    for(int i=2;i<=n;i++)
    {
        Log[i]=Log[i/2]+1;
    }
    for(int i=1;i<=n;i++)
    {
        f[0][i]=a[i];
    }
    for(int j=1;j<=Log[n];j++)
    {
        for(int i=1;i+(1<<j)-1<=n;i++)
        {
            f[j][i]=max(f[j-1][i],f[j-1][i+(1<<(j-1))]);
        }
    }
}

int cx(int l, int r)
{
    int len = r - l + 1;
    int y = Log[len];
    return max(f[y][l],f[y][r-(1<<y)+1]);
}

int main()
{
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++)
    {
        scanf("%d",&a[i]);
    }
    build(n);
    for(int i=1;i<=m;i++)
    {
        int x,y;
        scanf("%d%d",&x,&y);
        int z;
        z=cx(x,y);
        printf("%d  \n",z);
    }
    return 0;
}
